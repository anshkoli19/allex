import pandas as pd
import matplotlib.pyplot as plt
from tkinter import Tk, filedialog
from sklearn.preprocessing import StandardScaler, LabelEncoder
from sklearn.decomposition import PCA
# Select CSV dataset
root = Tk()
root.withdraw()
file_path = filedialog.askopenfilename(
    title="Select PCA Dataset",
    filetypes=[("CSV Files", "*.csv")]
)
root.destroy()
if not file_path:
    print("No dataset selected!")
    raise SystemExit
# Load dataset
df = pd.read_csv(file_path)
print("\nDataset Preview:")
print(df.head())
print("\nAvailable Columns:")
print(list(df.columns))
# Separate target column (optional)
target = input(
    "\nEnter target column (e.g. Species), or press Enter to skip: "
).strip()
if target and target not in df.columns:
    print("Error: Target column not found!")
    raise SystemExit

if target:
    labels = df[target].astype(str)
    features = df.drop(columns=[target])
else:
    labels = None
    features = df.copy()
# Keep numeric features only
features = features.select_dtypes(include="number")
features = features.dropna()
if features.shape[1] < 2:
    print("Error: PCA needs at least 2 numeric columns!")
    raise SystemExit
if len(features) < 3:
    print("Error: Dataset needs at least 3 valid rows!")
    raise SystemExit
# Match labels to rows retained after dropna
if labels is not None:
    labels = labels.loc[features.index]
# Standardize data
scaler = StandardScaler()
scaled_data = scaler.fit_transform(features)
# Apply PCA: reduce dimensions to 2
pca = PCA(n_components=2)
principal_components = pca.fit_transform(scaled_data)
# Results
result = pd.DataFrame(
    principal_components,
    columns=["PC1", "PC2"],
    index=features.index
)

if labels is not None:
    result["Species"] = labels
print("\nPCA Result (first 10 rows):")
print(result.head(10).round(3))
print("\nExplained Variance Ratio:")
print(pca.explained_variance_ratio_.round(4))
print(
    "Total Variance Explained:",
    round(pca.explained_variance_ratio_.sum() * 100, 2),
    "%"
)
# Plot PCA graph
plt.figure(figsize=(8, 6))
if labels is not None:
    for species in labels.unique():
        idx = labels == species
        plt.scatter(
            result.loc[idx, "PC1"],
            result.loc[idx, "PC2"],
            label=species
        )
    plt.legend()
else:
    plt.scatter(result["PC1"], result["PC2"])
plt.xlabel("Principal Component 1 (PC1)")
plt.ylabel("Principal Component 2 (PC2)")
plt.title("PCA - Dimensionality Reduction")
plt.grid(True)
plt.tight_layout()
plt.show()

# Save PCA output
result.to_csv("pca_output.csv", index=False)
print("\nPCA output saved as pca_output.csv")
