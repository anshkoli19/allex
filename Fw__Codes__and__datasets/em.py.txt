import pandas as pd
import matplotlib.pyplot as plt
from tkinter import Tk, filedialog
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler
from sklearn.mixture import GaussianMixture
from sklearn.decomposition import PCA
# Select dataset
root = Tk()
root.withdraw()
path = filedialog.askopenfilename(filetypes=[("CSV Files", "*.csv")])
root.destroy()
if not path:
    print("No dataset selected!")
    raise SystemExit
# Load and prepare data
df = pd.read_csv(path)
X = df.select_dtypes(include="number")
X = X.replace([float("inf"), float("-inf")], float("nan"))
X = X.dropna(axis=1, how="all")
if len(X) < 3 or X.shape[1] == 0:
    print("Dataset needs 3+ rows and numeric columns!")
    raise SystemExit
X = SimpleImputer(strategy="median").fit_transform(X)
X = pd.DataFrame(X).loc[:, pd.DataFrame(X).nunique() > 1].values
if X.shape[1] == 0:
    print("No usable numeric features!")
    raise SystemExit

X = StandardScaler().fit_transform(X)
# Expectation Maximization
k = min(3, len(X) - 1)
model = GaussianMixture(n_components=k, random_state=42)
clusters = model.fit_predict(X)
print("Clusters:", clusters)
print("Converged:", model.converged_)
print("Iterations:", model.n_iter_)
# Visualize clusters
points = PCA(n_components=2).fit_transform(X) if X.shape[1] >= 2 else X
if points.shape[1] == 1:
    points = __import__("numpy").column_stack([points[:, 0], [0] * len(points)])
plt.scatter(points[:, 0], points[:, 1], c=clusters)
plt.title("Expectation Maximization Clustering")
plt.xlabel("Component 1")
plt.ylabel("Component 2")
plt.show()
# Save output
df["Cluster"] = clusters
df.to_csv("em_output.csv", index=False)
print("Saved: em_output.csv")
